cmake_minimum_required(VERSION 3.24)
project(SLXResilienceLab VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 OpenGL Svg Network SerialPort)
set(_private_default OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/private/session.cpp")
    set(_private_default ON)
endif()
option(SLX_BUILD_PRIVATE_CORE "Build proprietary implementation (internal repository only)" ${_private_default})
set(SLX_CORE_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.ResilienceLab.Core.dll.a" CACHE FILEPATH "Matching compiled core import library")
if(SLX_BUILD_PRIVATE_CORE)
    add_library(slx-resilience-core SHARED private/session.cpp private/simulation/plant.cpp private/communication/bus.cpp private/control/controller.cpp private/control/arbiter.cpp private/faults/faults.cpp private/recording/recorder.cpp)
    target_include_directories(slx-resilience-core PUBLIC core/include PRIVATE private)
    target_compile_definitions(slx-resilience-core PRIVATE SLX_CORE_BUILD QT_NO_KEYWORDS)
    target_link_libraries(slx-resilience-core PUBLIC Qt6::Core)
    set_target_properties(slx-resilience-core PROPERTIES OUTPUT_NAME "SiliconLogiX.ResilienceLab.Core" PREFIX "" IMPORT_PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
else()
    if(NOT EXISTS "${SLX_CORE_BINARY}")
        message(FATAL_ERROR "SLX_CORE_BINARY must point to an approved, ABI-compatible core library")
    endif()
    if(SLX_CORE_BINARY STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.ResilienceLab.Core.dll.a")
        if(NOT WIN32 OR NOT MINGW OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "13.1.0" OR NOT Qt6_VERSION VERSION_EQUAL "6.12.0")
            message(FATAL_ERROR "Included SDK requires Windows x64 / Qt 6.12.0 / MinGW GCC 13.1.0; obtain a matching SDK for other kits")
        endif()
    endif()
    add_library(slx-resilience-core INTERFACE)
    target_include_directories(slx-resilience-core INTERFACE core/include)
    target_link_libraries(slx-resilience-core INTERFACE "${SLX_CORE_BINARY}" Qt6::Core)
endif()
qt_add_executable(slx-resiliencelab src/main.cpp src/labbridge.cpp src/labbridge.h src/benchapi.cpp src/benchapi.h src/simulationpacer.cpp src/campaign.cpp src/valveview.cpp src/valveview.h src/report.cpp src/report.h resources.qrc)
target_link_libraries(slx-resiliencelab PRIVATE slx-resilience-core Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL Qt6::Svg Qt6::Network Qt6::SerialPort)
target_compile_definitions(slx-resiliencelab PRIVATE SLX_APP_VERSION="${PROJECT_VERSION}")
set_target_properties(slx-resiliencelab PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
if(WIN32)
    qt_add_executable(slx-icon-maker EXCLUDE_FROM_ALL scripts/make-icon.cpp)
    target_link_libraries(slx-icon-maker PRIVATE Qt6::Gui Qt6::Svg)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/brand/app-icon.ico")
        target_sources(slx-resiliencelab PRIVATE assets/brand/appicon.rc)
    endif()
endif()
if(NOT SLX_BUILD_PRIVATE_CORE AND WIN32)
    add_custom_command(TARGET slx-resiliencelab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.ResilienceLab.Core.dll" "$<TARGET_FILE_DIR:slx-resiliencelab>")
endif()
add_library(slx-customer-example SHARED examples/customer-controller/controller.cpp)
target_include_directories(slx-customer-example PRIVATE core/include)
set_target_properties(slx-customer-example PROPERTIES PREFIX "" OUTPUT_NAME "SLX.CustomerController.Example")
foreach(target slx-resiliencelab slx-customer-example)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
    endif()
endforeach()
if(SLX_BUILD_PRIVATE_CORE AND NOT MSVC)
    target_compile_options(slx-resilience-core PRIVATE -Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
include(CTest)
if(BUILD_TESTING AND SLX_BUILD_PRIVATE_CORE)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(core-tests tests/core_tests.cpp)
    target_include_directories(core-tests PRIVATE private)
    target_link_libraries(core-tests PRIVATE slx-resilience-core Qt6::Test)
    target_compile_definitions(core-tests PRIVATE SLX_EXAMPLE_PLUGIN="$<TARGET_FILE:slx-customer-example>")
    add_dependencies(core-tests slx-customer-example)
    add_test(NAME resilience-contract COMMAND core-tests)
    qt_add_executable(protocol-tests tests/protocol_tests.cpp private/communication/bus.cpp private/control/controller.cpp private/control/arbiter.cpp)
    target_include_directories(protocol-tests PRIVATE private core/include)
    target_compile_definitions(protocol-tests PRIVATE QT_NO_KEYWORDS)
    target_link_libraries(protocol-tests PRIVATE Qt6::Test)
    add_test(NAME message-arbitration COMMAND protocol-tests)
    qt_add_executable(integration-tests tests/integration_tests.cpp src/labbridge.cpp src/labbridge.h src/simulationpacer.cpp src/campaign.cpp src/report.cpp)
    target_include_directories(integration-tests PRIVATE src)
    target_link_libraries(integration-tests PRIVATE slx-resilience-core Qt6::Test)
    add_dependencies(integration-tests slx-customer-example)
    add_test(NAME scenario-report-integration COMMAND integration-tests)
    qt_add_executable(desktop-tests tests/desktop_tests.cpp src/labbridge.cpp src/labbridge.h src/simulationpacer.cpp src/campaign.cpp src/valveview.cpp src/valveview.h src/report.cpp resources.qrc)
    target_include_directories(desktop-tests PRIVATE src)
    target_link_libraries(desktop-tests PRIVATE slx-resilience-core Qt6::Test Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL Qt6::Svg)
    add_dependencies(desktop-tests slx-customer-example)
    add_test(NAME desktop-interaction COMMAND desktop-tests)
    set_tests_properties(desktop-interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(bench-api-tests tests/bench_api_tests.cpp src/benchapi.cpp src/benchapi.h src/labbridge.cpp src/labbridge.h src/simulationpacer.cpp src/campaign.cpp src/report.cpp)
    target_include_directories(bench-api-tests PRIVATE src)
    target_link_libraries(bench-api-tests PRIVATE slx-resilience-core Qt6::Test Qt6::Network Qt6::SerialPort)
    add_test(NAME live-bench-api COMMAND bench-api-tests)
endif()
if(BUILD_TESTING)
    add_test(NAME desktop-smoke COMMAND slx-resiliencelab --smoke-test)
    set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
